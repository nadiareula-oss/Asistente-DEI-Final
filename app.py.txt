import streamlit as st
import google.generativeai as genai
from pathlib import Path

# --- 1. CONFIGURACIÓN INICIAL ---
st.set_page_config(page_title="Asistente de Revisión DEI", page_icon="💬")

# --- 2. CARGA DEL PROMPT DESDE EL ARCHIVO ---
@st.cache_data
def load_prompt():
    try:
        return Path("prompt.txt").read_text(encoding="utf-8")
    except FileNotFoundError:
        st.error("Error: No se encontró el archivo 'prompt.txt'. Asegúrate de que esté subido a tu repositorio de GitHub.")
        st.stop()

SYSTEM_PROMPT = load_prompt()

# --- 3. CONFIGURACIÓN DEL MODELO Y SECRETS ---
try:
    genai.configure(api_key=st.secrets["GOOGLE_API_KEY"])
except Exception:
    st.error("Error al configurar la API de Google. Asegúrate de haber añadido tu clave de API a los secretos de Streamlit.")
    st.stop()

model = genai.GenerativeModel(
    model_name='gemini-1.5-pro-latest',
    system_instruction=SYSTEM_PROMPT
)

# --- 4. INTERFAZ DE LA APLICACIÓN ---
st.title("Asistente de Revisión DEI 💬")
st.write("¡Hola! Soy tu asistente de comunicación inclusiva. Pegá el texto que deseás revisar y te daré sugerencias basadas en nuestras Guías de Diversidad, Equidad e Inclusión.")

if "chat" not in st.session_state:
    st.session_state.chat = model.start_chat(history=[])

for message in st.session_state.chat.history:
    role = "Tú" if message.role == "user" else "Asistente"
    with st.chat_message(role):
        st.markdown(message.parts[0].text)

if prompt := st.chat_input("Pega el texto que deseas revisar aquí..."):
    with st.chat_message("Tú"):
        st.markdown(prompt)

    with st.spinner("Analizando..."):
        response = st.session_state.chat.send_message(prompt)
        with st.chat_message("Asistente"):
            st.markdown(response.text)
